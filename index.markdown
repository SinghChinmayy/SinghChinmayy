/**
 * Updates the time display with IST and local time
 */
function updateTime() {
  const now = new Date();
  
  // IST time using timezone
  const istTimeString = now.toLocaleString('en-US', {
    timeZone: 'Asia/Kolkata',
    hour12: true,
    hour: '2-digit',
    minute: '2-digit'
  });
  
  // Local time
  const localTimeString = now.toLocaleTimeString('en-US', {
    hour12: true,
    hour: '2-digit',
    minute: '2-digit'
  });
  
  // Update DOM elements
  document.getElementById('ist-time').textContent = istTimeString;
  document.getElementById('local-time').textContent = localTimeString;
}

// Initialize time display
updateTime();

// Update time every minute
setInterval(updateTime, 60000);
</script>
