<script>
function updateTime() {
  const now = new Date();
  
  // IST time - using toLocaleString with timezone
  const istTimeString = now.toLocaleString('en-US', {
    timeZone: 'Asia/Kolkata',
    hour12: true,
    hour: '2-digit',
    minute: '2-digit'
  });
  
  // Local time
  const localTimeString = now.toLocaleTimeString('en-US', {
    hour12: true,
    hour: '2-digit',
    minute: '2-digit'
  });
  
  document.getElementById('ist-time').textContent = istTimeString;
  document.getElementById('local-time').textContent = localTimeString;
}

// Update time immediately and then every minute
updateTime();
setInterval(updateTime, 60000);
